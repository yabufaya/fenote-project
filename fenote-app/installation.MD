# 1. Create the Vite React project
npm create vite@latest fenote-app -- --template react


@already vite is configured so install packages below
# 2. Enter the folder
cd fenote-app

# 3. Install core dependencies with exact versions
npm install react@18.2.0 react-dom@18.2.0

# 4. Install Tailwind and PostCSS with exact versions
npm install -D tailwindcss@3.4.1 postcss@8.4.35 autoprefixer@10.4.17

# 5. Install i18n with exact versions
npm install i18next@23.8.2 react-i18next@14.0.5 i18next-browser-languagedetector@7.2.0

# 6. Install Icons
npm install @iconify/react@4.1.1

# 7. Install Three.js and React Three Fiber (3D Background)
npm install three@0.161.0 @react-three/fiber@8 @react-three/drei@9.96.5
